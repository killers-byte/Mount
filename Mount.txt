-- ============================================================
-- STATE
-- ============================================================
local State = {
    SpeedEnabled = false,
    SpeedValue = 50,
    Smoothness = 0.15, -- makin kecil makin instan, makin besar makin halus
    FPSCap = 120,
    AntiLag = true,
    WalkSpeedBypass = true,
}

-- ============================================================
-- 1. AUTO FPS LOCK 120 (SAFE, MOBILE FRIENDLY)
-- ============================================================
local function applyFpsCap(cap)
    pcall(function()
        if setfpscap then
            setfpscap(cap)
        end
    end)
    -- fallback via fflag kalau setfpscap nggak ada
    pcall(function()
        if setfflag then
            setfflag("DFIntTaskSchedulerTargetFps", tostring(cap))
        end
    end)
end

applyFpsCap(State.FPSCap)

-- ============================================================
-- 2. ANTI-LAG: MATIKAN VISUAL BERAT (OPSIONAL, BISA DIMATIKAN)
-- ============================================================
local function optimizeVisuals()
    if not State.AntiLag then return end

    pcall(function()
        Lighting.GlobalShadows = false
        Lighting.FogStart = 0
        Lighting.FogEnd = 1e9
        Lighting.Brightness = 1
        Lighting.EnvironmentDiffuseScale = 0
        Lighting.EnvironmentSpecularScale = 0
    end)

    -- matikan post-processing
    pcall(function()
        for _, v in ipairs(Lighting:GetChildren()) do
            if v:IsA("PostEffect") then
                v.Enabled = false
            end
        end
    end)

    -- optimasi part di workspace secara bertahap (anti crash)
    task.spawn(function()
        local processed = 0
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("BasePart") then
                obj.Material = Enum.Material.Plastic
                obj.Reflectance = 0
                obj.CastShadow = false
            elseif obj:IsA("Decal") or obj:IsA("Texture") then
                obj.Transparency = 1
            elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail")
                or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
                obj.Enabled = false
            end
            processed += 1
            if processed % 250 == 0 then
                task.wait() -- yield biar nggak freeze
            end
        end
    end)
end

optimizeVisuals()

-- ============================================================
-- 3. BYPASS WALKSPEED (ANTI-CHEAT PROPERTY CHECK)
-- ============================================================
local oldIndex, oldNamecall
pcall(function()
    local mt = getrawmetatable(game)
    oldIndex = mt.__index
    oldNamecall = mt.__namecall
    setreadonly(mt, false)

    mt.__index = newcclosure(function(self, key)
        if State.WalkSpeedBypass and not checkcaller() then
            if key == "WalkSpeed" or key == "JumpPower" then
                return 16 -- nilai palsu yang dianggap legal
            end
        end
        return oldIndex(self, key)
    end)

    setreadonly(mt, true)
end)

-- ============================================================
-- 4. CHARACTER HANDLER
-- ============================================================
local Character, Humanoid, RootPart

local function onCharacterAdded(char)
    Character = char
    Humanoid = char:WaitForChild("Humanoid")
    RootPart = char:WaitForChild("HumanoidRootPart")

    if State.SpeedEnabled then
        Humanoid.WalkSpeed = State.SpeedValue
    end
end

LocalPlayer.CharacterAdded:Connect(onCharacterAdded)
if LocalPlayer.Character then
    onCharacterAdded(LocalPlayer.Character)
end

-- ============================================================
-- 5. SMOOTH SPEED ENGINE
-- ============================================================
-- Menggunakan TweenService agar transisi kecepatan halus, bukan loncat.
-- WalkSpeed di-tween dari nilai sekarang ke target dalam durasi Smoothness.
-- Ini menghindari deteksi "speed spike" yang sering jadi trigger anti-cheat.

local TweenService = game:GetService("TweenService")
local currentTween

local function applySpeedSmooth()
    if not Humanoid then return end

    if currentTween then
        currentTween:Cancel()
        currentTween = nil
    end

    if not State.SpeedEnabled then
        -- balik ke normal dengan halus
        local tweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        currentTween = TweenService:Create(Humanoid, tweenInfo, {WalkSpeed = 16})
        currentTween:Play()
        return
    end

    -- tween ke target speed
    local tweenInfo = TweenInfo.new(
        State.Smoothness,
        Enum.EasingStyle.Quad,
        Enum.EasingDirection.Out
    )
    currentTween = TweenService:Create(Humanoid, tweenInfo, {WalkSpeed = State.SpeedValue})
    currentTween:Play()
end

-- Jika user mengubah slider saat speed aktif, langsung tween ke nilai baru
local function setSpeed(value)
    State.SpeedValue = value
    if State.SpeedEnabled then
        applySpeedSmooth()
    end
end
